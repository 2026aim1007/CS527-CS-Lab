% Vector Addition of Two Arrays
% Assuming N is stored at address 0x0 and is a multiple of 8

x1 = 0
x2 = [x1]       % Load N (Total number of elements)

% Calculate start address of Array A (ptrA)
x3 = 4          % ptrA = 0x4

% Calculate start address of Array B (ptrB)
x4 = x2 * 4     % byte_offset = N * 4
x4 = x4 + 4     % ptrB = 4 + 4N

% Calculate start address of Array C / Result (ptrC)
x5 = x2 * 8     % byte_offset = N * 8
x5 = x5 + 4     % ptrC = 4 + 8N

.loop
x10 = x2 - 0    % Check if remaining elements to process (N) == 0
BEQ .exit       % If N == 0, exit loop

% Perform Vector Math
v1 = [x3]       % Load 8 elements from Array A (auto-increments x3 by 32)
v2 = [x4]       % Load 8 elements from Array B (auto-increments x4 by 32)
v3 = v1 + v2    % Vector Addition
[x5] = v3       % Store 8 elements to Array C (auto-increments x5 by 32)

x2 = x2 - 8     % Decrement remaining elements counter by 8
BAL .loop       % Unconditional jump back to loop

.exit
