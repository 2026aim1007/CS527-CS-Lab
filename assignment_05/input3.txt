% FIR Filter Implementation
% N stored at 0x0. Weights at 0x4. Input at 0x24 (36). Output at 0x100 (256).

x1 = 0
x2 = [x1]       % Load N (Total input size)
x2 = x2 - 8     % We need to run the loop while k < N-8

x10 = 4         % Address of weights
v1 = [x10]      % Load the 8 weights into v1 (x10 auto-increments to 36, but we won't use it)

x3 = 36         % ptrInput = 36 (0x24)
x4 = 256        % ptrOutput = 256 (0x100)
x5 = 0          % Loop counter (k = 0)

.loop
x11 = x5 - x2   % Compare k with N-8
BEQ .exit       % If k == N-8, we are done

x6 = x3         % Copy ptrInput to x6 because vector load will auto-increment it
v2 = [x6]       % Load input[k] to input[k+7] into v2
v3 = v1 * v2    % Vector multiply: input[k+i] * weight[i]

% Store the multiplied vector to scratchpad memory to sum the elements
x7 = 224        % Scratchpad memory address (0xE0)
[x7] = v3       % Write the 8 multiplied elements to memory

% Sum the 8 elements using scalar reads
x8 = 0          % sum = 0
x9 = [224]
x8 = x8 + x9
x9 = [228]
x8 = x8 + x9
x9 = [232]
x8 = x8 + x9
x9 = [236]
x8 = x8 + x9
x9 = [240]
x8 = x8 + x9
x9 = [244]
x8 = x8 + x9
x9 = [248]
x8 = x8 + x9
x9 = [252]
x8 = x8 + x9

% Store the final sum to the output array
x12 = x4        % Copy ptrOutput to avoid auto-incrementing issues with vector stores
[x12] = x8      % output[k] = sum
x4 = x4 + 4     % Increment ptrOutput by 4 bytes (1 element)

x3 = x3 + 4     % Increment ptrInput by 4 bytes (shift sliding window by 1)
x5 = x5 + 1     % k++
BAL .loop

.exit
